# invoice_agent.py — automated invoice processing
from meshon import Agent, guardrails
 
agent = Agent(
name="invoice-triage",
tools=[read_pdf, match_po, notify_team],
approvals="human-in-the-loop",
)
 
@agent.task("process invoice")
async def process(invoice: Invoice) -> Result:
data = await read_pdf(invoice.file)
order = await match_po(data.po_number)
if guardrails.amount_ok(order, data):
await notify_team(order, status="approved")
return Result(order, data, score=data.confidence)
